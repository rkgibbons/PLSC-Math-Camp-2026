\begin{frame}{The Good Stuff}
    To describe the \emph{possible set} of multiple characteristics, we take the \underline{outer product} of the relevant dimensions:
    
    \begin{align*}
        &\{\text{Char. }1\} \times \{\text{Char. }2\} \times ... \times \{\text{Char. }n\} \\
        = & \prod_{i = 1}^{n}\{\text{Char. }i\}
    \end{align*}

    This means that all observations will be vectors of the form:  

    \begin{align*}
        (\text{Char. }1, \text{Char. }2, ..., \text{Char. }n)
    \end{align*}
\end{frame}

\begin{frame}{Example: Location}
    We use latitude and longitude to describe geographic location:

    \begin{align*}
        \text{Earth} &:=  \{Longitude\} \times \{Latitude\} \\
        &:= \{x \in \mathbb{R}: -90 \leq x \leq 90\} \times \{y \in \mathbb{R}: |y| \leq 90\}
    \end{align*}

    Then, any location on earth can be exactly identified as some vector $(x,y)$.  \\
    \pause

    \emph{\textcolor{CadetBlue}{We live in 3-space. What assumption makes a 2-space measurement sufficient?}}
\end{frame}

\vspace{.3cm}
    Beyond an object's position, we could consider density, temperature, color, favorite album...

\begin{frame}{A Not-So-Scary Formulation}
     \begin{block}{Real Space}
        Suppose we want to identify $n$ characteristics. Then, we generally use $\mathbb{R} \times \mathbb{R} \times ... \times \mathbb{R}$. This is our construction of $\mathbb{R}^n$: ``real space."
    \end{block}

    \begin{block}{Vectors}
    A \textbf{vector} is an element of $\mathbb{R}^n$ (at least in this characterization).
    \end{block}
\end{frame}

\begin{frame}{A Brief Question}
    Why use $\mathbb{R}^n$ so generally? Why not $\mathbb{N}^n$ for entirely discrete data, or $\mathbb{R}^\ell \times \mathbb{N}^k$ for a combination of $\ell$ continuous values and $k$ discrete ones? 
\end{frame}

\begin{frame}{Measuring Vectors}

Suppose $\textbf{x} \in \mathbb{R}^n$ is a vector such that $\textbf{x} = (x_1,x_2,...,x_n)$.

We can characterize the \emph{length} or \emph{norm} of $\textbf{x}$ as its distance from \textbf{0}:

\begin{align*}
    ||\textbf{x}|| &= \Big(x_1^2 + x_2^2 + ... + x_n^2 \Big)^{1/2}
\end{align*}

(How is this connected to our absolute value operator?)

\end{frame}

\begin{frame}{Metrics}
    A \textbf{metric} is a function that returns the distance between two elements in a set.
\vspace{.5cm}
    We commonly refer to a metric as ``$d$'' such that $d(a,b) \in \mathbb{R}_+$
\vspace{.5cm}
    Though we can choose metrics rather freely, there are three conditions every metric must satisfy.
\end{frame}

\begin{frame}{Triangle Inequality}

The Triangle Inequality is a key foundation of algebra, and states that for any elements $x,y,z$:

\begin{align*}
    d(x,z) & \leq d(x,y) + d(y,z)
\end{align*}

That is -- no path between any two elements can be shorter than the ``direct'' route.

\end{frame}

\begin{frame}{Triangle Violations}
This seems rather obvious, but we can break it relatively easily:
\vspace{1cm}
Suppose $d(x,y) = (x - y)^2$. Does this hold for $(x,y,z) = (0,1,2)$?
\end{frame}

\begin{frame}{Symmetry}
Symmetry is perhaps our simplest condition: distance is equivalent in both directions.
\vspace{.7cm}

A metric $d$ is symmetric iff $d(x,y) = d(y,x)$ for all $x,y$.
\vspace{.7cm}

Suppose $d(x,y) = x-y$. Then, $d(2,1) = 1$, but $d(1,2) = -1$. So, symmetry is violated.
\end{frame}


\begin{frame}{Positive Definiteness}
A metric $d$ is \textbf{positive definite} iff it satisfies two conditions:

\begin{enumerate}
    \item For all $x,y$, we have $d(x,y) \geq 0$;
    \item $d(x,y) = 0 \iff x = y$.
\end{enumerate}

That is: no negative distances, and non-equivalency must return non-zero distance.

\end{frame}

\begin{frame}{The Discrete Metric}
A very simple metric we often consider is called the \textbf{discrete metric}, which returns a binary equivalency value:
\begin{align*}
    d(x,y) &= \begin{cases}
    1, & x \neq y \\
    0, & x = y
    \end{cases}
\end{align*}

Is this a valid metric? Is it useful in the social sciences?

\end{frame}

\begin{frame}{More Realistically...}

We have a standard ``default'' metric that we call \textbf{euclidean distance}:

\begin{align*}
    d(\textbf{x},\textbf{y}) &= \sqrt{(x_1 - y_1)^2 + (x_2 - y_2)^2 + ... + (x_n - y_n)^2} \\
    &= \Bigg(\bigg(\sum_{i=1}^n \Big| x_i - y_i \Big|^2 \bigg) \Bigg)^{1/2}
\end{align*}

\end{frame}

\begin{frame}{A Generalized Approach}

    Euclidean distance is one iteration of what we call the ``$p$-metric'':

    \begin{align*}
        d_p(\textbf{x},\textbf{y}) &= \Bigg(\bigg(\sum_{i=1}^n \Big| x_i - y_i \Big|^p \bigg) \Bigg)^{1/p}
    \end{align*}

    In the social sciences, if we want to use $p \neq 2$, we generally need to justify why...

\end{frame}
